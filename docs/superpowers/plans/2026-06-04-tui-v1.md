# TUI V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add the first useful Textual TUI for browsing persisted recommendation runs, inspecting recommendation details and score breakdowns, applying basic filters, switching runs, and recording feedback keybindings with local rerank.

**Architecture:** Keep the TUI offline-first by reading only DuckDB recommendation tables while the user navigates. Put query/filter/run/detail logic in a small data module, then build a Textual app around that data store with a recommendation list, detail pane, score breakdown, filter state, run picker, and feedback actions. Feedback keybindings delegate to the existing feedback-loop API, write events immediately, rerank the persisted run, then reload the local rows.

**Tech Stack:** Python 3.11+, DuckDB, Textual, Typer, Rich, pytest, ruff, mypy.

---

## Scope

This plan covers the TUI V1 slice from `docs/spec.md`:

- Section 16: persisted-run TUI, main recommendations/detail layout, keybindings, recommendation columns, and explanation detail.
- Section 17: feedback writes immediately and lightly reranks the current persisted run.
- Section 18.4: missing enrichment is represented in the TUI as clear local status labels.

This plan deliberately keeps these pieces out of scope:

- Live API refresh or enrichment from inside the TUI.
- Seeds, Tags, Feedback history management, and Doctor screens beyond status labels needed in the detail pane.
- Undo feedback. The `u` binding is reserved and shows a local "Undo is outside TUI V1" status message in this slice.
- Visual snapshot testing. Behavioral Textual tests with `App.run_test()` and `Pilot.press()` are enough for V1.

Prerequisites before executing this plan:

- The feedback-loop plan must be complete, including `src/scrobble_recs/models/feedback.py` with `record_feedback_event()` and `rerank_recommendation_run()`.
- Recommendation rows must contain inspectable explanation JSON as implemented by the recommender plans.

```python
from __future__ import annotations

import json
from dataclasses import dataclass
from typing import Any

from duckdb import DuckDBPyConnection


@dataclass(frozen=True)
class RecommendationFilters:
    search: str = ""
    tag: str = ""
    max_knownness: float | None = None
    max_obviousness: float | None = None


@dataclass(frozen=True)
class RunSummary:
    run_id: str
    created_at: str
    mode: str
    recommendations: int


@dataclass(frozen=True)
class RecommendationListRow:
    run_id: str
    artist_id: int
    rank: int
    artist: str
    score: float
    primary_tags: tuple[str, ...]
    knownness: float
    obviousness: float
    why_short: str
    feedback_state: str


@dataclass(frozen=True)
class ScoreBreakdown:
    final_score: float
    raw_score: float
    lastfm_similarity_score: float
    tag_similarity_score: float
    cluster_fit_score: float
    rediscovery_score: float
    metadata_graph_score: float
    feedback_score: float
    knownness: float
    obviousness: float
    diversity_penalty: float


@dataclass(frozen=True)
class RecommendationDetail:
    run_id: str
    artist_id: int
    artist: str
    rank: int
    summary: str
    seed_artists: tuple[str, ...]
    tags: tuple[str, ...]
    penalties: dict[str, float]
    classification: dict[str, bool]
    score_breakdown: ScoreBreakdown
    status_labels: tuple[str, ...]
    feedback_history: tuple[str, ...]


class TuiRecommendationStore:
    def __init__(self, conn: DuckDBPyConnection) -> None:
        self._conn = conn

    def list_runs(self) -> list[RunSummary]:
        rows = self._conn.execute(
            """
            SELECT
                CAST(rr.run_id AS VARCHAR),
                CAST(rr.created_at AS VARCHAR),
                rr.mode,
                count(r.artist_id)
            FROM recommendation_run rr
            LEFT JOIN recommendation r ON r.run_id = rr.run_id
            GROUP BY rr.run_id, rr.created_at, rr.mode
            ORDER BY rr.created_at DESC, rr.run_id DESC
            """
        ).fetchall()
        return [
            RunSummary(
                run_id=str(run_id),
                created_at=str(created_at),
                mode=str(mode),
                recommendations=int(count),
            )
            for run_id, created_at, mode, count in rows
        ]

    def latest_run_id(self) -> str:
        runs = self.list_runs()
        if not runs:
            raise ValueError("no recommendation runs exist")
        return runs[0].run_id

    def list_recommendations(
        self,
        filters: RecommendationFilters | None = None,
        *,
        run_id: str | None = None,
    ) -> list[RecommendationListRow]:
        resolved_run_id = run_id or self.latest_run_id()
        resolved_filters = filters or RecommendationFilters()
        rows = self._conn.execute(
            """
            SELECT
                CAST(r.run_id AS VARCHAR),
                r.artist_id,
                r.rank,
                a.canonical_name,
                r.final_score,
                r.knownness,
                r.obviousness,
                r.feedback_score,
                r.explanation
            FROM recommendation r
            JOIN artist a ON a.artist_id = r.artist_id
            WHERE r.run_id = ?
            ORDER BY r.rank
            """,
            [resolved_run_id],
        ).fetchall()
        recommendations = [
            _list_row(
                run_id=str(row[0]),
                artist_id=int(row[1]),
                rank=int(row[2]),
                artist=str(row[3]),
                final_score=float(row[4] or 0.0),
                knownness=float(row[5] or 0.0),
                obviousness=float(row[6] or 0.0),
                feedback_score=float(row[7] or 0.0),
                explanation=_parse_json(row[8]),
            )
            for row in rows
        ]
        return [
            row
            for row in recommendations
            if _matches_filters(row, resolved_filters)
        ]

    def get_detail(self, run_id: str, *, artist_id: int) -> RecommendationDetail:
        row = self._conn.execute(
            """
            SELECT
                r.rank,
                a.canonical_name,
                r.final_score,
                r.raw_score,
                r.lastfm_similarity_score,
                r.tag_similarity_score,
                r.cluster_fit_score,
                r.rediscovery_score,
                r.metadata_graph_score,
                r.feedback_score,
                r.knownness,
                r.obviousness,
                r.diversity_penalty,
                r.explanation
            FROM recommendation r
            JOIN artist a ON a.artist_id = r.artist_id
            WHERE r.run_id = ?
              AND r.artist_id = ?
            """,
            [run_id, artist_id],
        ).fetchone()
        if row is None:
            raise ValueError(f"recommendation {artist_id} does not exist in run {run_id}")
        explanation = _parse_json(row[13])
        score_breakdown = ScoreBreakdown(
            final_score=float(row[2] or 0.0),
            raw_score=float(row[3] or 0.0),
            lastfm_similarity_score=float(row[4] or 0.0),
            tag_similarity_score=float(row[5] or 0.0),
            cluster_fit_score=float(row[6] or 0.0),
            rediscovery_score=float(row[7] or 0.0),
            metadata_graph_score=float(row[8] or 0.0),
            feedback_score=float(row[9] or 0.0),
            knownness=float(row[10] or 0.0),
            obviousness=float(row[11] or 0.0),
            diversity_penalty=float(row[12] or 0.0),
        )
        return RecommendationDetail(
            run_id=run_id,
            artist_id=artist_id,
            artist=str(row[1]),
            rank=int(row[0]),
            summary=str(explanation.get("summary", "")),
            seed_artists=_seed_names(explanation),
            tags=_tag_names(explanation),
            penalties=_float_map(explanation.get("penalties", {})),
            classification=_bool_map(explanation.get("classification", {})),
            score_breakdown=score_breakdown,
            status_labels=_status_labels(explanation, score_breakdown),
            feedback_history=self.feedback_history(run_id=run_id, artist_id=artist_id),
        )

    def feedback_history(self, *, run_id: str, artist_id: int) -> tuple[str, ...]:
        table_exists = self._conn.execute(
            """
            SELECT count(*)
            FROM information_schema.tables
            WHERE table_schema = 'main'
              AND table_name = 'feedback_event'
            """
        ).fetchone()
        if table_exists == (0,):
            return ()
        rows = self._conn.execute(
            """
            SELECT action
            FROM feedback_event
            WHERE run_id = ?
              AND artist_id = ?
            ORDER BY created_at, feedback_id
            """,
            [run_id, artist_id],
        ).fetchall()
        return tuple(str(row[0]) for row in rows)


def _list_row(
    *,
    run_id: str,
    artist_id: int,
    rank: int,
    artist: str,
    final_score: float,
    knownness: float,
    obviousness: float,
    feedback_score: float,
    explanation: dict[str, Any],
) -> RecommendationListRow:
    return RecommendationListRow(
        run_id=run_id,
        artist_id=artist_id,
        rank=rank,
        artist=artist,
        score=final_score,
        primary_tags=_tag_names(explanation)[:3],
        knownness=knownness,
        obviousness=obviousness,
        why_short=str(explanation.get("summary", "")),
        feedback_state=_feedback_state(feedback_score),
    )


def _matches_filters(row: RecommendationListRow, filters: RecommendationFilters) -> bool:
    search = filters.search.strip().casefold()
    tag = filters.tag.strip().casefold()
    if search and search not in row.artist.casefold() and search not in row.why_short.casefold():
        return False
    if tag and tag not in {item.casefold() for item in row.primary_tags}:
        return False
    if filters.max_knownness is not None and row.knownness > filters.max_knownness:
        return False
    if filters.max_obviousness is not None and row.obviousness > filters.max_obviousness:
        return False
    return True


def _parse_json(value: object) -> dict[str, Any]:
    if value is None:
        return {}
    if isinstance(value, dict):
        return value
    parsed = json.loads(str(value))
    return parsed if isinstance(parsed, dict) else {}


def _tag_names(explanation: dict[str, Any]) -> tuple[str, ...]:
    tags = explanation.get("tags", [])
    if not isinstance(tags, list):
        return ()
    names = []
    for item in tags:
        if isinstance(item, dict):
            tag = str(item.get("tag", "")).strip()
            if tag:
                names.append(tag)
    return tuple(names)


def _seed_names(explanation: dict[str, Any]) -> tuple[str, ...]:
    seeds = explanation.get("seed_artists", [])
    if not isinstance(seeds, list):
        return ()
    names = []
    for item in seeds:
        if isinstance(item, dict):
            artist = str(item.get("artist", "")).strip()
            if artist:
                names.append(artist)
    return tuple(names)


def _float_map(value: object) -> dict[str, float]:
    if not isinstance(value, dict):
        return {}
    return {str(key): float(raw or 0.0) for key, raw in value.items()}


def _bool_map(value: object) -> dict[str, bool]:
    if not isinstance(value, dict):
        return {}
    return {str(key): bool(raw) for key, raw in value.items()}


def _status_labels(
    explanation: dict[str, Any],
    score_breakdown: ScoreBreakdown,
) -> tuple[str, ...]:
    labels = []
    if not _seed_names(explanation) or score_breakdown.lastfm_similarity_score == 0.0:
        labels.append("partial_similarity")
    if not _tag_names(explanation) or score_breakdown.tag_similarity_score == 0.0:
        labels.append("partial_tags")
    return tuple(labels) or ("complete",)


def _feedback_state(feedback_score: float) -> str:
    if feedback_score > 0.0:
        return "boosted"
    if feedback_score < 0.0:
        return "suppressed"
    return ""

from __future__ import annotations

from typing import ClassVar

from duckdb import DuckDBPyConnection
from rich.text import Text
from textual.app import App, ComposeResult
from textual.binding import Binding
from textual.containers import Horizontal, Vertical
from textual.widgets import Footer, Header, Static

from scrobble_recs.tui.data import (
    RecommendationDetail,
    RecommendationFilters,
    RecommendationListRow,
    TuiRecommendationStore,
)


class ScrobbleRecsTui(App[None]):
    CSS = """
    Screen {
        layout: vertical;
    }

    #body {
        height: 1fr;
    }

    #recommendations {
        width: 44%;
        border: solid $accent;
        padding: 0 1;
    }

    #right-pane {
        width: 56%;
        height: 1fr;
    }

    #detail {
        height: 2fr;
        border: solid $accent;
        padding: 0 1;
    }

    #score-breakdown {
        height: 1fr;
        border: solid $accent;
        padding: 0 1;
    }

    #status {
        height: 1;
        dock: bottom;
        padding: 0 1;
    }
    """

    BINDINGS: ClassVar = [
        Binding("q", "quit", "Quit"),
        Binding("j", "move_down", "Down"),
        Binding("k", "move_up", "Up"),
        Binding("s", "feedback('save')", "Save"),
        Binding("r", "feedback('reject')", "Reject"),
        Binding("a", "feedback('already_know')", "Know"),
        Binding("o", "feedback('too_obvious')", "Obvious"),
        Binding("w", "feedback('wrong_genre')", "Wrong genre"),
        Binding("m", "feedback('more_like_this')", "More"),
        Binding("l", "feedback('less_like_this')", "Less"),
        Binding("h", "feedback('hide_artist')", "Hide"),
        Binding("u", "notify_undo_unavailable", "Undo"),
        Binding("?", "help", "Help"),
    ]

    def __init__(
        self,
        conn: DuckDBPyConnection,
        *,
        store: TuiRecommendationStore | None = None,
    ) -> None:
        super().__init__()
        self.conn = conn
        self.store = store or TuiRecommendationStore(conn)
        self.filters = RecommendationFilters()
        self.run_id: str | None = None
        self.rows: list[RecommendationListRow] = []
        self.selected_index = 0
        self.selected_artist_id: int | None = None
        self.show_debug_scores = False

    def compose(self) -> ComposeResult:
        yield Header()
        with Horizontal(id="body"):
            yield Static(id="recommendations")
            with Vertical(id="right-pane"):
                yield Static(id="detail")
                yield Static(id="score-breakdown")
        yield Static(id="status")
        yield Footer()

    def on_mount(self) -> None:
        try:
            self.run_id = self.store.latest_run_id()
        except ValueError as exc:
            self.query_one("#status", Static).update(str(exc))
            return
        self._reload_rows()

    def action_move_down(self) -> None:
        if not self.rows:
            return
        self.selected_index = min(self.selected_index + 1, len(self.rows) - 1)
        self._render_all()

    def action_move_up(self) -> None:
        if not self.rows:
            return
        self.selected_index = max(self.selected_index - 1, 0)
        self._render_all()

    def action_notify_undo_unavailable(self) -> None:
        self.query_one("#status", Static).update("Undo is outside TUI V1")

    def action_help(self) -> None:
        self.query_one("#status", Static).update(
            "Keys: j/k move, s save, r reject, a know, o obvious, m more, l less, h hide, R run"
        )

    def action_feedback(self, action: str) -> None:
        self.query_one("#status", Static).update(f"Feedback action queued: {action}")

    def _reload_rows(self) -> None:
        if self.run_id is None:
            return
        self.rows = self.store.list_recommendations(self.filters, run_id=self.run_id)
        if self.selected_index >= len(self.rows):
            self.selected_index = max(len(self.rows) - 1, 0)
        self._render_all()

    def _render_all(self) -> None:
        self.query_one("#recommendations", Static).update(self._recommendation_text())
        detail = self._selected_detail()
        self.selected_artist_id = detail.artist_id if detail else None
        self.query_one("#detail", Static).update(self._detail_text(detail))
        self.query_one("#score-breakdown", Static).update(self._score_text(detail))
        self.query_one("#status", Static).update(self._status_text())

    def _selected_detail(self) -> RecommendationDetail | None:
        if self.run_id is None or not self.rows:
            return None
        selected = self.rows[self.selected_index]
        return self.store.get_detail(self.run_id, artist_id=selected.artist_id)

    def _recommendation_text(self) -> Text:
        text = Text()
        text.append("Recommendations\n", style="bold")
        for index, row in enumerate(self.rows):
            marker = ">" if index == self.selected_index else " "
            tags = ", ".join(row.primary_tags)
            text.append(
                f"{marker} {row.rank:>3}. {row.artist:<28} {row.score:0.3f} {tags}\n"
            )
        if not self.rows:
            text.append("No recommendations match the current filters.\n")
        return text

    def _detail_text(self, detail: RecommendationDetail | None) -> Text:
        if detail is None:
            return Text("No recommendation selected")
        text = Text()
        text.append(f"{detail.artist}\n", style="bold")
        text.append(f"Rank: {detail.rank}\n")
        text.append(f"Why: {detail.summary}\n")
        if detail.seed_artists:
            text.append(f"Seeds: {', '.join(detail.seed_artists)}\n")
        if detail.tags:
            text.append(f"Tags: {', '.join(detail.tags)}\n")
        text.append(f"Status: {', '.join(detail.status_labels)}\n")
        if detail.feedback_history:
            text.append(f"Feedback: {', '.join(detail.feedback_history)}\n")
        return text

    def _score_text(self, detail: RecommendationDetail | None) -> Text:
        if detail is None:
            return Text("No scores")
        score = detail.score_breakdown
        text = Text()
        text.append("Score breakdown\n", style="bold")
        text.append(f"final: {score.final_score:0.3f}\n")
        text.append(f"raw: {score.raw_score:0.3f}\n")
        text.append(f"knownness: {score.knownness:0.3f}\n")
        text.append(f"obviousness: {score.obviousness:0.3f}\n")
        text.append(f"feedback: {score.feedback_score:0.3f}\n")
        if self.show_debug_scores:
            text.append(f"lastfm: {score.lastfm_similarity_score:0.3f}\n")
            text.append(f"tag: {score.tag_similarity_score:0.3f}\n")
            text.append(f"cluster: {score.cluster_fit_score:0.3f}\n")
            text.append(f"rediscovery: {score.rediscovery_score:0.3f}\n")
            text.append(f"metadata: {score.metadata_graph_score:0.3f}\n")
            text.append(f"diversity penalty: {score.diversity_penalty:0.3f}\n")
        return text

    def _status_text(self) -> str:
        run = self.run_id or "none"
        return (
            f"run: {run} | filter search={self.filters.search or '*'} "
            f"tag={self.filters.tag or '*'} | rows: {len(self.rows)}"
        )

            return
        tags = list(detail.tags)
        try:
            record_feedback_event(
                self.conn,
                run_id=self.run_id,
                artist_id=detail.artist_id,
                action=action,
                context={"tags": tags},
            )
            stats = rerank_recommendation_run(self.conn, self.run_id)
        except ValueError as exc:
            self.query_one("#status", Static).update(f"Feedback error: {exc}")
            return
        selected_artist_id = detail.artist_id
        self.rows = self.store.list_recommendations(self.filters, run_id=self.run_id)
        self.selected_index = _index_for_artist(self.rows, selected_artist_id)
        self._render_all()
        self.query_one("#status", Static).update(
            f"Feedback recorded: {action}; reranked {stats.recommendations}; hidden {stats.hidden}"
        )
```

def _index_for_artist(rows: list[RecommendationListRow], artist_id: int) -> int:
    for index, row in enumerate(rows):
